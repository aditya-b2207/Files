USE SalesOrderUAT;
GO

SELECT DISTINCT
    pm.CompanyCode,
    ps.State AS SourceState,
    pm.Plant AS SourcePlant,

    srm.StateName AS DestinationState,
    pm.Region AS DestinationRegionCode,

    pm.Material_Number AS MaterialCode,
    pm.Material_Description AS ProductName,

    pm.Condition_Table,
    pm.Condition_Type,
    pm.Condition_Amount,

    pm.Valid_From,
    pm.Valid_To

FROM dbo.TB_PriceMaster pm

INNER JOIN dbo.TB_Plant_State ps
    ON LTRIM(RTRIM(pm.Plant))
       = LTRIM(RTRIM(ps.Plant))

INNER JOIN dbo.TB_StateRegionMap srm
    ON LTRIM(RTRIM(pm.Region))
       = LTRIM(RTRIM(srm.RegionCode))

WHERE pm.CompanyCode = '1054'

    AND pm.Condition_Table = 'A706'

    AND pm.Condition_Type = 'ZB01'

    -- Source Location Punjab
    AND UPPER(LTRIM(RTRIM(ps.State))) = 'PUNJAB'

    -- Exclude Punjab as Destination
    AND UPPER(LTRIM(RTRIM(srm.StateName))) <> 'PUNJAB'

    -- Currently Valid Prices
    AND (
        pm.Valid_From IS NULL
        OR pm.Valid_From <= CAST(GETDATE() AS DATE)
    )

    AND (
        pm.Valid_To IS NULL
        OR pm.Valid_To >= CAST(GETDATE() AS DATE)
    )

ORDER BY
    DestinationRegionCode,
    DestinationState,
    ProductName;
