USE SalesOrderUAT;
GO

SELECT
    pm.Customer_Number AS CustomerCode,
    pm.SalesOrganisation,
    pm.Material_Number,
    pm.Material_Description,
    pm.Condition_Amount,
    pm.Condition_Unit,
    pm.Unit_Of_Measurement,
    pm.Plant,
    pm.Region,
    pm.Valid_From,
    pm.Valid_To,
    pm.Condition_Table,
    pm.Condition_Type
FROM dbo.TB_PriceMaster pm
WHERE pm.CompanyCode = '1054'
  AND pm.Condition_Table = 'A860'
  AND pm.Condition_Type = 'ZB01'
  AND LTRIM(RTRIM(pm.Customer_Number)) = 'MPOT004009'
  AND (
      pm.Valid_From IS NULL
      OR pm.Valid_From <= CAST(GETDATE() AS DATE)
  )
  AND (
      pm.Valid_To IS NULL
      OR pm.Valid_To >= CAST(GETDATE() AS DATE)
  )
ORDER BY pm.Id DESC;
