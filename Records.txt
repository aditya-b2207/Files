SELECT
    CreateOrderNumber,
    '[' + Plant + ']' AS OrderPlant,
    LEN(Plant) AS PlantLength,
    DATALENGTH(Plant) AS PlantBytes
FROM TB_Create_Dealer_Order
WHERE CreateOrderNumber = 5204;

-----------------------------------------------------------------
SELECT
    PlantCode,
    '[' + PlantCode + ']' AS PlantCodeCheck,
    EmailID
FROM TB_CFA_List
WHERE PlantCode LIKE '%G100%';

---------------------------------------------------------------------------------
SELECT *
FROM TB_CFA_List
WHERE PlantCode = 'G100';

-----------------------------------------------------------------------------

DECLARE @CreateOrderNumber BIGINT = 5204;

SELECT
    O.Plant AS OrderPlant,
    C.PlantCode,
    C.EmailID
FROM TB_Create_Dealer_Order O
LEFT JOIN TB_CFA_List C
    ON C.PlantCode = O.Plant
WHERE O.CreateOrderNumber = @CreateOrderNumber;
