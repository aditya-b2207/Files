        U.UserId,
        U.UserCode,
        U.Name AS UserName,
        U.RepManCode,
        U.RoleId,
        U.IsActive AS CurrentIsActive
    FROM @DealerList D

    INNER JOIN dbo.CustomerMasterData CM
        ON CM.CustomerCode = D.SoldToParty
       AND CM.CompanyCode = 1054

    INNER JOIN dbo.Users U
        ON U.UserCode = D.SoldToParty
       AND U.CompanyCode = 1054
       AND U.RoleId = 1

    ORDER BY
        D.DealerType,
        D.SoldToParty;


    ------------------------------------------------------------
    -- 4. CHECK EXCEL RECORDS PRESENT IN CUSTOMER MASTER

        CM.CustomerCode,
        CM.CustomerName,
        CM.TM
    FROM @DealerList D

    INNER JOIN dbo.CustomerMasterData CM
        ON CM.CustomerCode = D.SoldToParty
       AND CM.CompanyCode = 1054

    LEFT JOIN dbo.Users U
        ON U.UserCode = D.SoldToParty
       AND U.CompanyCode = 1054

    -- Currently inactive/null
    ------------------------------------------------------------

    UPDATE U
    SET
        U.IsActive = 1,
        U.Updated_At = GETDATE()
    FROM dbo.Users U

    INNER JOIN @DealerList D
        ON D.SoldToParty = U.UserCode

    INNER JOIN dbo.CustomerMasterData CM
        ON CM.CustomerCode = D.SoldToParty
       AND CM.CompanyCode = 1054

    WHERE U.CompanyCode = 1054
      AND U.RoleId = 1
      AND ISNULL(U.IsActive, 0) = 0;
