CREATE TABLE dbo.TB_SmtpDiagnosticLog
(
    SmtpDiagnosticLogId BIGINT IDENTITY(1,1) PRIMARY KEY,

    CreateOrderNumber BIGINT NULL,
    SONumber NVARCHAR(100) NULL,
    EventType NVARCHAR(100) NULL,

    Stage NVARCHAR(50) NULL,
    Status NVARCHAR(50) NULL,

    SmtpHost NVARCHAR(255) NULL,
    SmtpPort INT NULL,

    FromEmail NVARCHAR(320) NULL,
    ToEmails NVARCHAR(MAX) NULL,
    CcEmails NVARCHAR(MAX) NULL,
    Subject NVARCHAR(500) NULL,

    SmtpResponse NVARCHAR(MAX) NULL,

    SmtpStatusCode NVARCHAR(100) NULL,
    SmtpErrorCode NVARCHAR(100) NULL,
    RejectedMailbox NVARCHAR(320) NULL,

    ExceptionType NVARCHAR(500) NULL,
    ErrorMessage NVARCHAR(MAX) NULL,
    InnerException NVARCHAR(MAX) NULL,

    Source NVARCHAR(50) NULL,
    Version NVARCHAR(100) NULL,

    LoggedOn DATETIME2(3) NOT NULL
        CONSTRAINT DF_TB_SmtpDiagnosticLog_LoggedOn
        DEFAULT SYSDATETIME()
);

-------------------------------------------------------------------------------------------------------------------------------------------------

CREATE INDEX IX_TB_SmtpDiagnosticLog_CreateOrderNumber
ON dbo.TB_SmtpDiagnosticLog
(
    CreateOrderNumber,
    LoggedOn DESC
);
