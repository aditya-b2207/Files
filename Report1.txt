USE SalesOrderUAT;
GO

SELECT
    pm.Customer_Number AS CustomerCode,
    pm.Material_Number AS MaterialCode,
    pm.Material_Description AS ProductName,

    pm.Condition_Amount AS Price,

    pm.Plant AS SourcePlant,
    pm.Region AS RegionCode,

    pm.Condition_Table,
    pm.Condition_Type,

    pm.Valid_From,
    pm.Valid_To

FROM dbo.TB_PriceMaster pm

WHERE pm.CompanyCode = '1054'
  AND pm.Condition_Table = 'A860'
  AND pm.Condition_Type = 'ZB01'

  AND (pm.Valid_From IS NULL
       OR pm.Valid_From <= CAST(GETDATE() AS DATE))

  AND (pm.Valid_To IS NULL
       OR pm.Valid_To >= CAST(GETDATE() AS DATE))

ORDER BY
    pm.Customer_Number,
    pm.Material_Description,
    pm.Id DESC;
