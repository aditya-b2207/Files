-- CFA Mail (TO)
SELECT DISTINCT
    TRIM(C.EmailID) AS EmailID
FROM TB_CFA_List C
WHERE TRIM(C.PlantCode) = (
    SELECT TOP 1 TRIM(O.Plant)
    FROM TB_Create_Dealer_Order O
    WHERE O.CreateOrderNumber = @CreateOrderNumber
)
AND C.EmailID IS NOT NULL
AND TRIM(C.EmailID) <> '';
